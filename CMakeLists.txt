cmake_minimum_required(VERSION 3.16)
project(MathOmega LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(mathomega_cpp cpp/main.cpp)
if(MSVC)
  target_compile_options(mathomega_cpp PRIVATE /W4)
else()
  target_compile_options(mathomega_cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()
